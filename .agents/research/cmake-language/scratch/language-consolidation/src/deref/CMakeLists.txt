cmake_minimum_required(VERSION 3.25)
project(deref LANGUAGES NONE)
# M1: unquoted ${A} whose value names another variable is dereferenced twice
set(A "B")
set(B "x")
if(${A} STREQUAL "x")
  message(STATUS "M1 unquoted: TRUE (double dereference)")
else()
  message(STATUS "M1 unquoted: FALSE")
endif()
if("${A}" STREQUAL "x")
  message(STATUS "M1 quoted: TRUE")
else()
  message(STATUS "M1 quoted: FALSE (value compared as written)")
endif()
# M2: a list value splits an unquoted operand into several if() arguments
set(L "a;b")
if("${L}" STREQUAL "a;b")
  message(STATUS "M2 quoted list: TRUE")
endif()
# M3: bare foreach drops empty elements; IN LISTS keeps them
set(E "a;;b")
set(n 0)
foreach(x ${E})
  math(EXPR n "${n}+1")
endforeach()
set(m 0)
foreach(x IN LISTS E)
  math(EXPR m "${m}+1")
endforeach()
message(STATUS "M3 bare foreach iterations=${n} IN LISTS iterations=${m}")
# M4: macro ARGN foreach drops empty arguments
macro(mac)
  set(k 0)
  foreach(x ${ARGN})
    math(EXPR k "${k}+1")
  endforeach()
  message(STATUS "M4 macro foreach over ARGN iterations=${k} of 3 passed")
endmacro()
mac(a "" b)
# M5: a generator expression is literal text at configure time
message(STATUS "M5 genex at configure: $<CONFIG>")
# M6: an empty unquoted operand is a hard error
set(EMPTY "")
if(DEFINED RUN_EMPTY)
  if(${EMPTY} STREQUAL "")
    message(STATUS "M6 unreachable")
  endif()
endif()
