cmake_minimum_required(VERSION 3.25)
project(parse LANGUAGES NONE)
# P1: PARSE_ARGN (4.4) - parses after the named parameters
if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.4 OR DEFINED FORCE_ARGN)
  function(fa name)
    cmake_parse_arguments(PARSE_ARGN arg "" "VALUE" "")
    message(STATUS "P1 PARSE_ARGN name=${name} VALUE='${arg_VALUE}' unparsed='${arg_UNPARSED_ARGUMENTS}'")
  endfunction()
  fa(n VALUE "a;b")
endif()
# P2: PARSE_ARGV inside a macro
if(DEFINED MACRO_ARGV)
  macro(ma)
    cmake_parse_arguments(PARSE_ARGV 0 arg "" "VALUE" "")
  endmacro()
  ma(VALUE x)
endif()
# P3: PARENT_SCOPE inside a macro targets the scope that contains the call
macro(set_out)
  set(OUT "from-macro" PARENT_SCOPE)
endmacro()
function(wrapper)
  set_out()
  message(STATUS "P3 inside wrapper after macro: OUT='${OUT}'")
endfunction()
function(caller)
  wrapper()
  message(STATUS "P3 in caller of wrapper: OUT='${OUT}'")
endfunction()
caller()
add_subdirectory(sub)
message(STATUS "P3 top dir after subdir called macro at dir scope: OUT='${OUT}'")
