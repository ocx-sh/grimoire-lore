cmake_minimum_required(VERSION 3.20)
project(pkgconfig_probe LANGUAGES NONE)

find_package(PkgConfig REQUIRED)
message(STATUS "ENV PKG_CONFIG_SYSROOT_DIR=$ENV{PKG_CONFIG_SYSROOT_DIR}")
message(STATUS "ENV PKG_CONFIG_LIBDIR=$ENV{PKG_CONFIG_LIBDIR}")
message(STATUS "ENV PKG_CONFIG_PATH=$ENV{PKG_CONFIG_PATH}")

pkg_check_modules(LIBFOO QUIET IMPORTED_TARGET libfoo)
if(LIBFOO_FOUND)
  message(STATUS "RESULT LIBFOO_FOUND=TRUE")
  message(STATUS "RESULT LIBFOO_INCLUDE_DIRS=${LIBFOO_INCLUDE_DIRS}")
  message(STATUS "RESULT LIBFOO_LIBRARY_DIRS=${LIBFOO_LIBRARY_DIRS}")
  message(STATUS "RESULT LIBFOO_LDFLAGS=${LIBFOO_LDFLAGS}")
else()
  message(STATUS "RESULT LIBFOO_FOUND=FALSE")
endif()
