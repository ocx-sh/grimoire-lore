cmake_minimum_required(VERSION 3.25...4.4)
project(dep09b LANGUAGES NONE)
include(FetchContent)
# SHAPE: plain | fpa | override | both
if(SHAPE STREQUAL "plain")
  FetchContent_Declare(dep SOURCE_DIR "${SRC}")
elseif(SHAPE STREQUAL "fpa")
  FetchContent_Declare(dep SOURCE_DIR "${SRC}" FIND_PACKAGE_ARGS 2.0)
elseif(SHAPE STREQUAL "override")
  FetchContent_Declare(dep SOURCE_DIR "${SRC}" OVERRIDE_FIND_PACKAGE)
else()
  FetchContent_Declare(dep SOURCE_DIR "${SRC}" OVERRIDE_FIND_PACKAGE FIND_PACKAGE_ARGS 2.0)
endif()
FetchContent_MakeAvailable(dep)
find_package(dep 3.0 EXACT REQUIRED)
message(STATUS "[dep09b] SHAPE=${SHAPE} found=${dep_FOUND} dep_VERSION='${dep_VERSION}'")
