cmake_minimum_required(VERSION 3.25...4.4)
project(p LANGUAGES C)
if(MODE STREQUAL "check")
  include(CheckCSourceCompiles)
  check_c_source_compiles("int main(void){return 0;}" HAVE_X)
elseif(MODE STREQUAL "sources")
  try_compile(S1 SOURCE_FROM_CONTENT a.c "int main(void){return 0;}")
  try_compile(S2 SOURCE_FROM_CONTENT b.c "int main(void){return 0;}")
elseif(MODE STREQUAL "whole")
  try_compile(W1 PROJECT w SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/whole BINARY_DIR ${CMAKE_BINARY_DIR}/w1)
  try_compile(W2 PROJECT w SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/whole BINARY_DIR ${CMAKE_BINARY_DIR}/w2)
endif()
message(STATUS "MODE=${MODE} S1=${S1} S2=${S2} W1=${W1} W2=${W2} HAVE_X=${HAVE_X}")
