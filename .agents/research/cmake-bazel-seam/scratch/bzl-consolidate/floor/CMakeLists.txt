cmake_minimum_required(VERSION 3.25...4.4)
project(floorprobe LANGUAGES C)
add_library(fp STATIC fp.c)
install(TARGETS fp EXPORT fpTargets ARCHIVE DESTINATION lib)
install(EXPORT fpTargets NAMESPACE fp:: DESTINATION lib/cmake/fp)
if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.3)
  install(PACKAGE_INFO fp EXPORT fpTargets)
  message(STATUS "CPS emitted")
else()
  message(STATUS "CPS skipped on ${CMAKE_VERSION}")
endif()
