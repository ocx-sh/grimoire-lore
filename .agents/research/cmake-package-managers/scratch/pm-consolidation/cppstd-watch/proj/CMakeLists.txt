cmake_minimum_required(VERSION 3.25)
project(cppstd_watch LANGUAGES CXX)
if(MODE STREQUAL "set20")
  set(CMAKE_CXX_STANDARD 20)
elseif(MODE STREQUAL "set17")
  set(CMAKE_CXX_STANDARD 17)
endif()
add_library(lib STATIC lib.cpp)
if(MODE STREQUAL "feature20")
  target_compile_features(lib PUBLIC cxx_std_20)
endif()
message(STATUS "MODE=${MODE} CMAKE_CXX_STANDARD=${CMAKE_CXX_STANDARD}")
