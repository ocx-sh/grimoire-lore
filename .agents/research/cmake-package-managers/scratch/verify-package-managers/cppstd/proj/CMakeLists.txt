cmake_minimum_required(VERSION 3.25)
project(p CXX)
if(MODE STREQUAL "set20")
  set(CMAKE_CXX_STANDARD 20)
elseif(MODE STREQUAL "guarded20")
  if(NOT DEFINED CMAKE_CXX_STANDARD)
    set(CMAKE_CXX_STANDARD 20)
  endif()
endif()
add_library(l STATIC lib.cpp)
if(MODE STREQUAL "feature20")
  target_compile_features(l PUBLIC cxx_std_20)
endif()
message(STATUS "MODE=${MODE} CMAKE_CXX_STANDARD=${CMAKE_CXX_STANDARD}")
